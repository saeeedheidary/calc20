(()=>{

/* ================= موتور محاسبه ================= */
let entry='0',stored=null,op=null,reset=false,memory='0',lastOp=null,lastVal=null,err=false;
const norm=s=>{if(!s||s==='-')return s||'0';if(s.includes('.'))s=s.replace(/0+$/,'').replace(/\.$/,'');return s==='-0'?'0':s};
function dec(s){s=String(s);let neg=s[0]==='-';if(neg)s=s.slice(1);let [a,b='']=s.split('.');return {n:BigInt((a||'0')+b)*(neg?-1n:1n),scale:b.length}}
function fmt(n,scale){let neg=n<0n;if(neg)n=-n;let s=n.toString().padStart(scale+1,'0');if(scale)s=s.slice(0,-scale)+'.'+s.slice(-scale);return norm((neg?'-':'')+s)}
function calc(a,b,o){
  let x=dec(a),y=dec(b),sc=Math.max(x.scale,y.scale),
      X=x.n*10n**BigInt(sc-x.scale),Y=y.n*10n**BigInt(sc-y.scale);
  if(o==='+')return fmt(X+Y,sc);
  if(o==='−')return fmt(X-Y,sc);
  if(o==='×')return fmt(x.n*y.n,x.scale+y.scale);
  if(o==='÷'){if(Y===0n)throw Error();let precision=12;return fmt((X*10n**BigInt(precision))/Y,sc+precision)}
  return b;
}
function shown(){
  let s=entry;let digits=s.replace(/[-.]/g,'').length;
  if(digits>20){s=s.slice(0,20+(s[0]==='-'?1:0));entry=s}
  const parts=entry.split('.');const negative=parts[0].startsWith('-');
  let whole=negative?parts[0].slice(1):parts[0];
  whole=whole.replace(/\B(?=(\d{3})+(?!\d))/g,',');
  const formatted=(negative?'-':'')+whole+(parts.length>1?'.'+parts.slice(1).join('.'):'');
  $('display').textContent=err?'خطا':formatted;
  $('history').textContent=op&&stored!==null?stored+' '+op:' ';
  $('mem').textContent='MEM '+(memory==='0'?'—':'●');
  $('digits').textContent=digits.toLocaleString('fa-IR')+' / ۲۰ رقم';
}
function input(n){
  if(err){entry='0';err=false}
  if(reset){entry=n==='.'?'0.':n;reset=false}
  else if(n==='.'){if(!entry.includes('.'))entry+='.'}
  else if(entry==='0')entry=n;
  else if(entry==='-0')entry='-'+n;
  else entry+=n;
  shown();
}
function choose(o){if(err)return;if(op&&!reset)equals();stored=entry;op=o;reset=true;shown()}
function equals(){
  if(op===null){if(lastOp){try{entry=calc(entry,lastVal,lastOp)}catch(e){err=true}}return shown()}
  try{let b=entry;entry=calc(stored,b,op);lastOp=op;lastVal=b;stored=null;op=null;reset=true}
  catch(e){err=true;entry='0'}
  shown();
}
function action(a){
  if(a==='clear'){entry='0';stored=null;op=null;lastOp=null;lastVal=null;reset=false;err=false}
  else if(a==='clearentry'){entry='0';err=false}
  else if(a==='back'){entry=entry.length>1?entry.slice(0,-1):'0';if(entry==='-')entry='0'}
  else if(a==='dot')input('.');
  else if(a==='plus'||a==='minus'||a==='multiply'||a==='divide')choose(({plus:'+',minus:'−',multiply:'×',divide:'÷'})[a]);
  else if(a==='equals')equals();
  else if(a==='sign'){if(entry!=='0')entry=entry[0]==='-'?entry.slice(1):'-'+entry}
  else if(a==='percent'){try{entry=fmt(dec(entry).n,dec(entry).scale+2)}catch(e){}}
  else if(a==='sqrt'){let v=Number(entry);if(v<0)err=true;else entry=String(Math.sqrt(v))}
  else if(a==='recip'){if(Number(entry)===0)err=true;else entry=String(1/Number(entry))}
  else if(a==='mc')memory='0';
  else if(a==='mr'){entry=memory;reset=true}
  else if(a==='mplus'||a==='mminus'){try{memory=calc(memory,entry,a==='mplus'?'+':'−')}catch(e){err=true}}
  else if(a==='repeat'){if(lastOp)equals()}
  shown();
}

});

function roundRect(g,a,b,w,h,r){
  g.beginPath();g.moveTo(a+r,b);g.lineTo(a+w-r,b);g.quadraticCurveTo(a+w,b,a+w,b+r);
  g.lineTo(a+w,b+h-r);g.quadraticCurveTo(a+w,b+h,a+w-r,b+h);g.lineTo(a+r,b+h);
  g.quadraticCurveTo(a,b+h,a,b+h-r);g.lineTo(a,b+r);g.quadraticCurveTo(a,b,a+r,b);g.closePath();
}

  return c.toDataURL('image/png');
}
